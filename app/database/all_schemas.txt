-- ─────────────────────────────────────────────────────────────────────────────
-- 1. LOGIN_DATA
-- ─────────────────────────────────────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS login_data (
    id                  INT AUTO_INCREMENT UNIQUE,
    email               VARCHAR(255) PRIMARY KEY,
    username            VARCHAR(50) NOT NULL,
    age                 TINYINT(3) NOT NULL,
    role                VARCHAR(20) NOT NULL,
    gender              VARCHAR(10) NOT NULL,
    password            VARCHAR(255) NOT NULL,
    otp                 VARCHAR(6) DEFAULT NULL,
    otp_created_at      DATETIME DEFAULT NULL,
    is_verified         TINYINT(1) DEFAULT 0,
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_email (email),
    INDEX idx_role (role)
);

-- 2. ENTREPRENEUR_PROFILE

-- 1. Store pre-computed embeddings for every user
--    Updated every time entrepreneur saves pitch or investor saves portfolio
CREATE TABLE IF NOT EXISTS user_embeddings (
    email        VARCHAR(255) PRIMARY KEY,
    role         ENUM('entrepreneur','investor') NOT NULL,
    embedding    MEDIUMTEXT NOT NULL,   -- JSON array, ~384 floats ≈ 7 KB
    updated_at   DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_ue_email FOREIGN KEY (email)
        REFERENCES login_data(email) ON DELETE CASCADE,
    INDEX idx_role (role)
);

-- 2. Cache of top-N match scores (avoids recomputing on every page load)
--    Direction: 'investor_to_entrepreneur' | 'entrepreneur_to_investor'
CREATE TABLE IF NOT EXISTS ai_match_cache (
    cache_id            INT AUTO_INCREMENT PRIMARY KEY,
    investor_email      VARCHAR(255) NOT NULL,
    entrepreneur_email  VARCHAR(255) NOT NULL,
    score               DECIMAL(6,4) NOT NULL,
    direction           ENUM('investor_to_entrepreneur','entrepreneur_to_investor') NOT NULL,
    computed_at         DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uq_match  (investor_email, entrepreneur_email, direction),
    INDEX idx_inv_dir    (investor_email,    direction, score DESC),
    INDEX idx_ent_dir    (entrepreneur_email, direction, score DESC),
    CONSTRAINT fk_amc_inv FOREIGN KEY (investor_email)
